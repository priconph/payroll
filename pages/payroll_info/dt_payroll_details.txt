        $tax_based = get_tax_annual($floated_tax_w_fcast_1);

        $rate_annl_earnings = $ytd_gross_income / ($ytd_gross_income + $forecast_earnings);

        $with_tax1 = $tax_based * $rate_annl_earnings;
        $with_tax = $with_tax1 - ($aRow['yt_tax_withheld'] + 3358.89); // for checking pa

        $row[] = number_format($with_tax,2); // With tax

        // Net pay
        $all_deductions = $eeshare + $phil_2_amnt + $pagibig_amnt + $sss_loan_amnt + $sss_cal_loan_amnt + $pagibig_sal_loan_amnt + $pagibig_cal_loan_amnt + $hipar_amnt + $adv_deduc_amnt + $tax_adj_amnt + $with_tax;
        $net_pay = $gross_pay_2 - $all_deductions; 
        $row[] = number_format($net_pay,2); // Net pay

		array_push($output['aaData'],$row);

	}

            if($floated_tax_w_fcast_1 < 250000){

    }

    function truncate_number( $number, $precision = 2){
        if ( 0 == (int)$number ) {
            return $number;
        }
        $negative = $number / abs($number);
        $number = abs($number);
        $precision = pow(10, $precision);
        return floor( $number * $precision ) / $precision * $negative;
    }
